------------------------- d -------------------------
-- d. Consulta que retorne o nome dos departamentos e calcule 
-- a quantidade de empregados e a média do salário dos empregados 
-- agrupado por departamento. 

db.empregado.aggregate([
    {
        $group: {
            _id: "$dno",
            quantidade_empregados: { $sum: 1},
            media_salarial: {$avg: "$salario"}
        }
    },

    {
        $lookup: {
            from: "departamento",
            localField: "_id",
            foreignField: "dnumero", 
            as: "departamento_info"
        }
    },

    {
        $unwind: "$departamento_info"
    },

    {
        $project:{
            nome_departamento: "$departamento_info.dnome", 
            quantidade_empregados: 1,
            media_salarial: {$round: ["$media_salarial", 2]},
            _id: 0
        }
    }
])

------------------------- e -------------------------
e. Consulta para recuperar os departamentos e a quantidade de empregados e 
-- a soma das horas trabalhadas por eles dos projetos do departamento, 
-- agrupado por departamento.

db.departamento.aggregate([
    // 1. lookup de empregados
    {
        $lookup: {
            from: "empregado",
            localField: "dnumero",
            foreignField: "dno", 
            as: "lista_empregados"
        }
    },

    // 2. Calcula a quantidade total de empregados
    {
        $addFields: {
            quantidade_empregados: { $size: "$lista_empregados" }
        }
    },

    // 3. Remover a lista de empregados
    {
        $project: { lista_empregados: 0 }
    },

    // 4. Projetos do departamento 
    {
        $lookup: {
            from: "projeto",
            localField: "dnumero",
            foreignField: "dnum",
            as: "lista_projetos"
        }
    },

    {
        $unwind: { path: "$lista_projetos", preserveNullAndEmptyArrays: true } // "true" minúsculo
    },

    // 5. Trabalha_em de cada projeto
    {
        $lookup: {
            from: "trabalha_em",
            localField: "lista_projetos.pnumero",
            foreignField: "pno",
            as: "trabalhos"
        }
    },

    {
        $unwind: { path: "$trabalhos", preserveNullAndEmptyArrays: true }
    },

    // 6 Agrupamento 
    {
        $group: {
            _id: "$dnumero",
            nome_departamento: { $first: "$dnome" }, 
            quantidade_empregados: { $first: "$quantidade_empregados" }, 
            soma_horas: { $sum: "$trabalhos.horas" }
        }
    }, 

    // 9. Projeção
    {
        $project: {
            _id: 0,
            nome_departamento: 1,
            quantidade_empregados: 1,
            soma_horas: 1
        }
    }
])

-- A figura abaixo mostra o log de execução de dois escalonamentos

------------------------- a -------------------------

--------------------- Resposta ----------------------
-- Após uma falha, transações que iniciaram mas não possuem um <commit> 
-- no log devem sofrer UNDO. Transações que possuem um <commit> no 
-- log devem sofrer REDO para garantir que suas alterações estejam no disco.

-- Undo: Não tem commit (antigo)
-- redo: Commit (atualizar)
-- Começa de baixo para cima a leitura

Lista-UNDO: T4, T5
Lista-REDO: T1, T2, T3

Resultado:
A: 27
B: 52
C: 42
D: 25
