-- Criar uma base de dados
CREATE DATABASE sistema_db;

-- Selecionar a base de dados para uso
USE sistema_db;

-- Criar pelo menos uma tabela com alguns campos
CREATE TABLE usuarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL,
    nivel_acesso ENUM('Admin', 'Usuario') DEFAULT 'Usuario',
    data_cadastro DATE
);

-- Inserir dados
INSERT INTO usuarios (nome, email, nivel_acesso, data_cadastro) 
VALUES ('Alan Turing', 'alan@computacao.com', 'Admin', '2026-05-25'),
       ('Ada Lovelace', 'ada@algoritmos.com', 'Usuario', '2026-05-25');

-- Alterar dados
UPDATE usuarios 
SET nivel_acesso = 'Admin' 
WHERE nome = 'Ada Lovelace';

-- Visualizar dados
SELECT * FROM usuarios;

-- Apagar dados
DELETE FROM usuarios 
WHERE id = 1;

-- Apagar tabela
DROP TABLE usuarios;

-- Apagar base de dados
DROP DATABASE sistema_db;
